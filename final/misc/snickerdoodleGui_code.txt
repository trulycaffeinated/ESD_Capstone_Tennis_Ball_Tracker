        function exportResultsToExcel(app)
            nFrame = numel(app.FrameHistory);
            nAct   = size(app.ActXYZHistory, 1);
            nCalc  = size(app.CalcXYZHistory, 1);
            nErr   = size(app.PctErrHistory, 1);
        
            n = min([nFrame, nAct, nCalc, nErr]);
        
            if n == 0
                warning("No valid calculated data available to export.");
                return;
            end
        
            if size(app.ActXYZHistory,2) < 3 || ...
               size(app.CalcXYZHistory,2) < 3 || ...
               size(app.PctErrHistory,2) < 3
                error("History arrays are not Nx3. Export aborted.");
            end
        
            T = table();
            T.Frame = app.FrameHistory(1:n)';
        
            % Actual
            T.Act_X = app.ActXYZHistory(1:n,1);
            T.Act_Y = app.ActXYZHistory(1:n,2);
            T.Act_Z = app.ActXYZHistory(1:n,3);
        
            % Calculated
            T.Calc_X = app.CalcXYZHistory(1:n,1);
            T.Calc_Y = app.CalcXYZHistory(1:n,2);
            T.Calc_Z = app.CalcXYZHistory(1:n,3);
        
            % Percent error
            T.PctErr_X = app.PctErrHistory(1:n,1);
            T.PctErr_Y = app.PctErrHistory(1:n,2);
            T.PctErr_Z = app.PctErrHistory(1:n,3);
        
            % Absolute error
            T.Err_X = T.Calc_X - T.Act_X;
            T.Err_Y = T.Calc_Y - T.Act_Y;
            T.Err_Z = T.Calc_Z - T.Act_Z;
        
            writetable(T, fullfile(pwd, 'triangulation_results.xlsx'));
        end

        function [u, v, dbg, center, radius] = grayscaleFindCentroid(app, img)
            %#ok<INUSD> % app not used here, kept for consistent signature
        
            grayImg = rgb2gray(img);
        
            eq = adapthisteq(grayImg);

            [centers, radii, metric] = imfindcircles(eq, ...
                [5,25], ...                   % radius search range
                'ObjectPolarity','bright',...   % detect bright circles
                'Sensitivity',0.95);              % detection sensitivity
        
            if isempty(centers)
                error('No circles detected (Grayscale).');
            end
        
            if radii < 13
                [centers, radii] = imfindcircles(eq, ...
                [5,25], ...                   % radius search range
                'ObjectPolarity','dark',...   % detect dark circles
                'Sensitivity',0.97);          % detection sensitivity
            end

            [~, i] = max(metric);
        
            center = centers(i, :);
            radius = radii(i);
        
            u = center(1);
            v = center(2);
        
            dbg = grayImg;
        end

        function [u, v, dbg, center, radius] = bwFindCentroid(app, img) 
            %#ok<INUSD> % app not used here, kept for consistent signature       
            grayImg = rgb2gray(img);
            bw = imbinarize(grayImg);
        
            % Clean up (tune these if needed)
            bw = bwareaopen(bw, 200);       % remove small specks
            bw = imfill(bw, 'holes');       % fill ball interior holes
        
            cc = bwconncomp(bw);
            if cc.NumObjects == 0
                error('No blobs detected (Binarize).');
            end
        
            % Largest blob assumed to be the ball
            numPixels = cellfun(@numel, cc.PixelIdxList);
            [~, idx] = max(numPixels);
        
            ballMask = false(size(bw));
            ballMask(cc.PixelIdxList{idx}) = true;
        
            stats = regionprops(ballMask, 'Centroid', 'Area');
            if isempty(stats)
                error('No centroid found (Binarize).');
            end
        
            center = stats.Centroid;      % [u v]
            u = center(1);
            v = center(2);
        
            radius = sqrt(stats.Area/pi); % equivalent radius from area
        
            dbg = ballMask;               % nice debug image
        end

        function [u, v, dbg, center, radius] = yCbCrFindCentroid(app, img)
            %#ok<INUSD> % app not used here, kept for consistent signature
        
            ycc = rgb2ycbcr(img);
            Y  = ycc(:,:,1);
            Cb = ycc(:,:,2);
            Cr = ycc(:,:,3);
        
            % Color threshold (tune for your tennis ball / lighting)
            colorMask = (Cr > 120) & (Cb < 120);
        
            % Clean mask
            mask = bwareaopen(colorMask, 200);
            mask = imfill(mask, 'holes');
        
            % Debug / detection image: keep only masked Y channel
            dbg = Y;
            dbg(~mask) = 0;
        
            [centers, radii, metric] = imfindcircles(dbg, [15 200], ...
                'ObjectPolarity', 'bright', ...
                'Sensitivity', 0.95);
        
            if isempty(centers)
                error('No circles detected (YCbCr).');
            end
        
            [~, i] = max(metric);
        
            center = centers(i, :);
            radius = radii(i);
        
            u = center(1);
            v = center(2);
        end

        function [u, v, dbg, center, radius] = hsvFindCentroid(app, img)
            %#ok<INUSD> % app not used directly, kept for consistent signature
        
            % Convert to HSV
            hsvImg = rgb2hsv(img);
            H = hsvImg(:, :, 1);
            S = hsvImg(:, :, 2);
            V = hsvImg(:, :, 3);
        
            % Thresholds for tennis ball color
            hueMask = (H > 0.18 & H < 0.20);
            satMask = (S > 0 & S < 1);
            valMask = (V > 0 & V < 1);
        
            % Final binary mask
            finalMask = hueMask & satMask & valMask;
        
            % Cleanup
            finalMask = bwareaopen(finalMask, 50);
            finalMask = imfill(finalMask, 'holes');
        
            % Fail if nothing found
            if ~any(finalMask(:))
                error('No valid HSV region found.');
            end
        
            %% REGION BOUNDING
            topRow    = find(any(finalMask, 2), 1, 'first');
            bottomRow = find(any(finalMask, 2), 1, 'last');
            leftCol   = find(any(finalMask, 1), 1, 'first');
            rightCol  = find(any(finalMask, 1), 1, 'last');
        
            if isempty(topRow) || isempty(bottomRow) || isempty(leftCol) || isempty(rightCol)
                error('No valid region found in HSV mask.');
            end
        
            % Crop mask to bounded region
            croppedMask = finalMask(topRow:bottomRow, leftCol:rightCol);
        
            % Region properties
            stats = regionprops(croppedMask, ...
                'Area', 'BoundingBox', 'Centroid', ...
                'MajorAxisLength', 'MinorAxisLength', 'Eccentricity');
        
            if isempty(stats)
                error('No centroid found in cropped HSV region.');
            end
        
            % Pick largest region if more than one exists
            [~, idx] = max([stats.Area]);
            statsMain = stats(idx);
        
            % Centroid in cropped coordinates
            c = statsMain.Centroid;
        
            % Convert centroid back to ROI image coordinates
            u = c(1) + leftCol - 1;
            v = c(2) + topRow - 1;
            center = [u, v];
        
            % Estimate radius from area
            radius = sqrt(statsMain.Area / pi);
        
            % Debug image: masked RGB image is nicer than raw HSV for display
            dbg = img;
            dbg(repmat(~finalMask, [1 1 3])) = 0;
        end

        function [z, x, y, d] = find3DCentroid(app, fpx, b, uL, vL, uR, cX, cY)
            %#ok<INUSD>  % app not used (suppresses warning)
            d = uL - uR; % disparity 

            z = (fpx * b) / d; % depth [mm]
            x = (uL - cX) * z / fpx; % x cord [mm]
            y = -(vL - cY) * z / fpx; % y cord [mm]
        end

        function Xw = triangulateDLT(app, PL, PR, uL, vL, uR, vR)
            %#ok<INUSD>
            A = [
                uL * PL(3,:) - PL(1,:);
                vL * PL(3,:) - PL(2,:);
                uR * PR(3,:) - PR(1,:);
                vR * PR(3,:) - PR(2,:);
            ];
        
            [~, ~, V] = svd(A, 0);
            Xh = V(:, end);
            Xh = Xh ./ Xh(4);
        
            Xw = Xh(1:3);   % world coordinates
        end

        function P = buildProjectionMatrix(app, camPos_m, Rwc_blender, W, H)
            %#ok<INUSD>
        
            % Intrinsics
            f_m  = 11e-3;
            px_m = 19e-6;
            fpx  = f_m / px_m;
            cX   = W / 2;
            cY   = H / 2;
        
            K = [fpx 0   cX;
                 0   fpx cY;
                 0   0   1];
        
            % Blender camera local axes -> CV camera axes
            % Blender local: +X right, +Y up, -Z forward
            % CV/slides:     +X right, +Y down, +Z forward
            Rfix = diag([1 -1 -1]);
        
            % world->camera rotation in CV convention
            Rcw = Rfix * Rwc_blender';
        
            % translation
            Cw = camPos_m(:);
            t = -Rcw * Cw;
        
            P = K * [Rcw t];
        end

        % Refactored applyBounding to be generic to allow me to use it on
        % either image individually
        function [imgROI, x0, y0] = applyBoundingGeneric(app, img, useBounding, topPct, bottomPct, leftPct, rightPct)
            if isempty(img)
                imgROI = img; x0 = 0; y0 = 0; return;
            end
        
            if ~useBounding
                imgROI = img; x0 = 0; y0 = 0; return;
            end
        
            H = size(img,1);
            W = size(img,2);
        
            % Clamp 0..100
            topPct    = max(0, min(topPct, 100));
            bottomPct = max(0, min(bottomPct, 100));
            leftPct   = max(0, min(leftPct, 100));
            rightPct  = max(0, min(rightPct, 100));
        
            if leftPct + rightPct >= 100 || topPct + bottomPct >= 100
                warning("Bounding box too tight: crops overlap. Reduce crop percentages.");
            end
        
            % Convert crop % -> pixel indices
            x1 = round(1 + (leftPct/100)   * (W-1));
            x2 = round(W - (rightPct/100) * (W-1));
            y1 = round(1 + (topPct/100)    * (H-1));
            y2 = round(H - (bottomPct/100)* (H-1));
        
            % Clamp bounds
            x1 = max(1, min(x1, W));  x2 = max(1, min(x2, W));
            y1 = max(1, min(y1, H));  y2 = max(1, min(y2, H));
        
            % Prevent collapse
            if x2 <= x1
                mid = round((x1 + x2)/2);
                x1 = max(1, mid-1);
                x2 = min(W, mid+1);
            end
            if y2 <= y1
                mid = round((y1 + y2)/2);
                y1 = max(1, mid-1);
                y2 = min(H, mid+1);
            end
        
            imgROI = img(y1:y2, x1:x2, :);
        
            x0 = x1 - 1;
            y0 = y1 - 1;
        end

        %% RUN LIVE CALCULATIONS
        function runLiveCalculation(app)
            if app.IsProcessing
                return;
            end
            app.IsProcessing = true;
            cleanupObj = onCleanup(@() releaseProcessing(app)); %#ok<NASGU>
        
            title(app.UIAxes3, sprintf("Running calc on frame %d", app.SetID));
            title(app.UIAxes4, sprintf("Running calc on frame %d", app.SetID));
            drawnow limitrate nocallbacks
        
            t0 = tic;
        
            leftIm  = getappdata(app.UIFigure, "LeftIm");
            rightIm = getappdata(app.UIFigure, "RightIm");
        
            if isempty(leftIm)
                uialert(app.UIFigure, 'Please load images first (Run Blender).', 'No images loaded');
                return;
            end

            method = app.ColorMethodDropDown.Value;

            switch method
                case 'Grayscale'
                    methodID = uint8(0);
                case 'Binarize (B&W)'
                    methodID = uint8(1);
                case 'YCbCr'
                    methodID = uint8(2);
                case 'HSV'
                    methodID = uint8(3);
                otherwise
                    methodID = uint8(0);
            end
            
            try
                [left_proc, right_proc, uL, vL, uR, vR] = snickerdoodleLink(app.snickClient, leftIm, rightIm, methodID);
            catch ME
                uialert(app.UIFigure, "Snickerdoodle Error: " + string(ME.message), "Processing Error");
                return;
            end

            % display processed images
            imshow(left_proc, 'Parent', app.UIAxes3);
            hold(app.UIAxes3, 'on');
            plot(app.UIAxes3, uL, vL, 'r+', 'LineWidth', 2);
            hold(app.UIAxes3, 'off');

            imshow(right_proc, 'Parent', app.UIAxes4);
            hold(app.UIAxes4, 'on');
            plot(app.UIAxes4, uR, vR, 'r+', 'LineWidth', 2);
            hold(app.UIAxes4, 'off');
        
            if isnumeric(app.ballLocation) && numel(app.ballLocation) >= 3
                ball_m = double(app.ballLocation);
                
                app.XCord_actual.Value = ball_m(1);
                app.YCord_actual.Value = ball_m(2);
                app.ZCord_actual.Value = ball_m(3);
            else
                app.XCord_actual.Value = NaN;
                app.YCord_actual.Value = NaN;
                app.ZCord_actual.Value = NaN;
            end
        
            % Full-image coordinates
            uL_full = uL;
            vL_full = vL;
            uR_full = uR;
            vR_full = vR;
        
            % GUI outputs
            app.XCord_LC_EF.Value = uL_full;
            app.YCord_LC_EF.Value = vL_full;
            app.XCord_RC_EF.Value = uR_full;
            app.YCord_RC_EF.Value = vR_full;
        
            H = size(leftIm,1);
            W = size(leftIm,2);
            app.CamRows_EF.Value = H;
            app.CamCol_EF.Value  = W;
        
            % Pose-based triangulation
            camL_m = double(app.camL_Location(:));
            camR_m = double(app.camR_Location(:));
        
            RwcL = app.camL_RotationMatrix;
            RwcR = app.camR_RotationMatrix;
        
            PL = buildProjectionMatrix(app, camL_m, RwcL, W, H);
            PR = buildProjectionMatrix(app, camR_m, RwcR, W, H);
        
            Xw_m = triangulateDLT(app, PL, PR, uL_full, vL_full, uR_full, vR_full);
            Xw_mm = 1000 * Xw_m;
        
            % Already in meters from triangulation
            x = Xw_m(1);
            y = Xw_m(2);
            z = Xw_m(3);
            d = uL_full - uR_full;
        
            app.FrameCounter = app.FrameCounter + 1;
        
            hasCalc = all(isfinite([x y z])) && isfinite(d);
            hasAct  = all(isfinite([app.XCord_actual.Value, app.YCord_actual.Value, app.ZCord_actual.Value]));
        
            if hasCalc && hasAct
                act = [app.XCord_actual.Value, app.YCord_actual.Value, app.ZCord_actual.Value];
                calc = [x, y, z];
        
                denom = abs(act);
                denom(denom < 1e-9) = NaN;
                pctErr = 100 * abs(calc - act) ./ denom;
        
                app.FrameHistory(end+1)    = app.FrameCounter;
                app.ActXYZHistory(end+1,:) = act;
                app.CalcXYZHistory(end+1,:)= calc;
                app.PctErrHistory(end+1,:) = pctErr;
                
                % Error cloud data
                err = app.CalcXYZHistory - app.ActXYZHistory;
                
                errX = err(:,1);
                errY = err(:,2);
                
                app.ErrorScatter.XData = errX;
                app.ErrorScatter.YData = errY;

                % Compute max absolute error for symmetric bounds
                maxErr = max(abs([errX; errY]));
                
                % Avoid zero-size axes
                if maxErr < 1e-6
                    maxErr = 1e-3;
                end
                
                % Add small margin (10%)
                margin = 1.1;
                lim = maxErr * margin;
                
                xlim(app.ErrorCloud_Axes, [-lim lim]);
                ylim(app.ErrorCloud_Axes, [-lim lim]);
        
                for i = 1:3
                    app.LineAct(i).XData  = app.FrameHistory;
                    app.LineAct(i).YData  = app.ActXYZHistory(:,i);
        
                    app.LineCalc(i).XData = app.FrameHistory;
                    app.LineCalc(i).YData = app.CalcXYZHistory(:,i);
                end
        
                for i = 1:3
                    app.LineErr(i).XData = app.FrameHistory;
                    app.LineErr(i).YData = app.PctErrHistory(:,i);
                end
        
                Nkeep = 100;
                if numel(app.FrameHistory) > Nkeep
                    xwin = [app.FrameHistory(end-Nkeep+1) app.FrameHistory(end)];
                else
                    xwin = [1 max(10, app.FrameHistory(end))];
                end
                xlim(app.ActualVsCal_Axes, xwin);
                xlim(app.PercentError_Axes, xwin);
            end
        
            app.XCord_3D_EF.Value = x;
            app.YCord_3D_EF.Value = y;
            app.ZCord_3D_EF.Value = z;
        
            drawnow limitrate nocallbacks
            app.RuntimemsEditField.Value = toc(t0) * 1000;
        end

        function [leftImg, rightImg] = blenderLinkStereo(app, client, width, height, camID, x, y, z, pitch, roll, yaw, objectName)
            %#ok<INUSD> camID currently unused
        
            floats = single([width, height, x, y, z, pitch, roll, yaw]);
            write(client, typecast(floats, 'uint8'));   % 32 bytes
        
            nameBytes = uint8(char(objectName));
            nameBytes(end+1) = 0; % null terminator helps
            write(client, [nameBytes, zeros(1, 1024 - numel(nameBytes), 'uint8')]);
        
            hdr = app.readExact(client, 8);
            sizes = typecast(uint8(hdr), 'uint32');
            leftSize  = double(sizes(1));
            rightSize = double(sizes(2));
        
            leftRaw  = app.readExact(client, leftSize);
            rightRaw = app.readExact(client, rightSize);
        
            % Object location (3 floats)
            locRaw = app.readExact(client, 12);
            loc = typecast(uint8(locRaw), 'single');
            app.ballLocation = double([loc(1), loc(2), loc(3)]);
        
            % Left camera location (3 floats)
            camL_loc_raw = app.readExact(client, 12);
            camL_loc = typecast(uint8(camL_loc_raw), 'single');
        
            % Left camera rotation matrix (9 floats)
            camL_R_raw = app.readExact(client, 36);
            camL_R = typecast(uint8(camL_R_raw), 'single');
        
            % Right camera location (3 floats)
            camR_loc_raw = app.readExact(client, 12);
            camR_loc = typecast(uint8(camR_loc_raw), 'single');
        
            % Right camera rotation matrix (9 floats)
            camR_R_raw = app.readExact(client, 36);
            camR_R = typecast(uint8(camR_R_raw), 'single');
        
            % Current-frame camera pose
            app.camL_Location = double([camL_loc(1), camL_loc(2), camL_loc(3)]);
            app.camR_Location = double([camR_loc(1), camR_loc(2), camR_loc(3)]);
        
            % Python sends row-major flattened 3x3, reshape then transpose for MATLAB
            app.camL_RotationMatrix = reshape(double(camL_R), [3,3])';
            app.camR_RotationMatrix = reshape(double(camR_R), [3,3])';
        
            % --- Update GUI fields for Left Camera ---
            app.LeftCam_XCord.Value = app.camL_Location(1);
            app.LeftCam_YCord.Value = app.camL_Location(2);
            app.LeftCam_ZCord.Value = app.camL_Location(3);
        
            % --- Update GUI fields for Right Camera ---
            app.RightCam_XCord.Value = app.camR_Location(1);
            app.RightCam_YCord.Value = app.camR_Location(2);
            app.RightCam_ZCord.Value = app.camR_Location(3);
        
            leftRGBA  = reshape(uint8(leftRaw),  [4, width, height]);
            rightRGBA = reshape(uint8(rightRaw), [4, width, height]);
        
            leftRGBA  = permute(leftRGBA,  [3 2 1]);
            rightRGBA = permute(rightRGBA, [3 2 1]);
        
            leftImg  = leftRGBA(:,:,1:3);
            rightImg = rightRGBA(:,:,1:3);
        
            leftImg = flipud(leftImg);
            rightImg = flipud(rightImg);
        end

        function data = readExact(app, client, nBytes)
        % Read exactly nBytes from tcpclient (blocks until received or timeout).
            data = uint8([]);
            while numel(data) < nBytes
                chunk = read(client, nBytes - numel(data), "uint8");
                if isempty(chunk)
                    error("Socket read returned empty before receiving all bytes.");
                end
                data = [data; chunk(:)]; %#ok<AGROW>
            end
        end
 

    end

        % Button pushed function: RunBlenderButton
        function RunBlenderButtonPushed(app, event)
             %#ok<INUSD>
            app.FrameCounter   = 0;
            app.FrameHistory   = [];
            app.CalcXYZHistory = [];
            app.ActXYZHistory  = [];
            app.PctErrHistory  = [];
        
            if ~isempty(app.LineAct) && all(isvalid(app.LineAct))
                for i = 1:3
                    app.LineAct(i).XData  = nan; app.LineAct(i).YData  = nan;
                    app.LineCalc(i).XData = nan; app.LineCalc(i).YData = nan;
                    app.LineErr(i).XData  = nan; app.LineErr(i).YData  = nan;
                end
            end
        
            if ~isempty(app.ErrorScatter) && isvalid(app.ErrorScatter)
                app.ErrorScatter.XData = nan;
                app.ErrorScatter.YData = nan;
            end
        
            try
                client = tcpclient(app.server_ip, app.server_port, "Timeout", 20);
            catch
                uialert(app.UIFigure, ...
                    'Ensure Blender server is running', ...
                    'Could not connect');
                return;
            end
        
            width = 1920;
            height = 1080;
        
            datFile = fullfile(pwd, 'datFiles', 'serve1.dat');
        
            try
                posData = readmatrix(datFile);
            catch ME
                uialert(app.UIFigure, "Could not read .dat file: " + ME.message, "File Error");
                return;
            end
        
            if isempty(posData) || size(posData,2) < 3
                uialert(app.UIFigure, ...
                    '.dat file must contain at least 3 numeric columns: [x y z]', ...
                    'Invalid File');
                return;
            end
        
            posData = posData(:,1:3);
            N = size(posData,1);
        
            app.ImageSets = cell(1, N);
            app.BallLocationHistory = zeros(N,3);
        
            app.LeftCamLocationHistory  = zeros(N,3);
            app.RightCamLocationHistory = zeros(N,3);
        
            app.LeftCamRotationMatrixHistory  = zeros(3,3,N);
            app.RightCamRotationMatrixHistory = zeros(3,3,N);
        
            saveFolder = fullfile(pwd, "captures");
            if ~exist(saveFolder, 'dir')
                mkdir(saveFolder);
            end
        
            % ---- CAPTURE ALL FRAMES ----
            for k = 1:5
                dx = posData(k,1);
                dz = posData(k,2);
                dy = posData(k,3);
        
                noiseFactor = 0;
                dx = dx + noiseFactor * randn();
                dy = dy + noiseFactor * randn();
                dz = dz + noiseFactor * randn();
        
                noiseFactorRot = 0;
                yaw   = noiseFactorRot * randn();
                pitch = noiseFactorRot * randn();
                roll  = noiseFactorRot * randn();
        
                try
                    [leftImg, rightImg] = app.blenderLinkStereo(client, width, height, ...
                        1, dx, dy, dz, pitch, roll, yaw, "tennisBall");
                catch ME
                    uialert(app.UIFigure, "Blender link failed: " + ME.message, "Capture error");
                    return;
                end
        
                app.ImageSets{k} = {leftImg, rightImg};
                app.BallLocationHistory(k,:) = app.ballLocation;
        
                app.LeftCamLocationHistory(k,:)  = app.camL_Location;
                app.RightCamLocationHistory(k,:) = app.camR_Location;
        
                app.LeftCamRotationMatrixHistory(:,:,k)  = app.camL_RotationMatrix;
                app.RightCamRotationMatrixHistory(:,:,k) = app.camR_RotationMatrix;
        
                leftPath  = fullfile(saveFolder, sprintf('leftImg_%d.jpeg', k));
                rightPath = fullfile(saveFolder, sprintf('rightImg_%d.jpeg', k));
        
                imwrite(leftImg, leftPath);
                imwrite(rightImg, rightPath);
                disp("Frame " + k);
            end

            disp("All frames are uploaded");
        
            % ---- FORCE CALCULATION ON EVERY FRAME ----
           % for k = 1:N
              %  app.SetID = k;
              %  showCurrentSet(app);   % this calls runLiveCalculation(app)
         %   end
        
            % Return to first frame for display
            app.SetID = 1;
            showCurrentSet(app);
        
            % ---- EXPORT ONLY AFTER HISTORIES EXIST ----
            try
                exportResultsToExcel(app);
            catch ME
                uialert(app.UIFigure, "Excel export failed: " + ME.message, "Export Error");
            end
        end

    % App creation and deletion
    methods (Access = public)

        % Construct app
        function app = gui

            % Create UIFigure and components
            createComponents(app)

            % Register the app with App Designer
            registerApp(app, app.UIFigure)

            % Execute the startup function
            runStartupFcn(app, @startupFcn)

            if nargout == 0
                clear app
            end
        end

        % Code that executes before app deletion
        function delete(app)

            % Delete UIFigure when app is deleted
            delete(app.UIFigure)
        end
